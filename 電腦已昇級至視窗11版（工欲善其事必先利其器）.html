function Get-DirectorySize {
    param([string]$Path)

    if (-not (Test-Path $Path)) {
        return $null
    }

    $sum = Get-ChildItem $Path -File -Recurse -Force |
        Measure-Object -Property Length -Sum

    [pscustomobject]@{
        Path = $Path
        SizeGB = [math]::Round(($sum.Sum / 1GB), 2)
        Files = $sum.Count
    }
}

$ErrorActionPreference = "SilentlyContinue"

function Remove-FilesSafely {
    param([string]$Path)

    if (-not (Test-Path $Path)) {
        Write-Host "不存在：$Path" -ForegroundColor DarkGray
        return
    }

    Write-Host "檢查：$Path" -ForegroundColor Cyan

    Get-ChildItem $Path -Force -Recurse |
        Remove-Item -Recurse -Force -WhatIf
}

if ($CleanUserTemp) {
    Remove-FilesSafely $env:TEMP
}

if ($CleanSystemTemp) {
    Remove-FilesSafely "C:\Windows\Temp"
}

if ($CleanBrowserCache) {
    $chrome = "$env:LOCALAPPDATA\Google\Chrome\User Data"
    $edge = "$env:LOCALAPPDATA\Microsoft\Edge\User Data"

    foreach ($base in @($chrome,$edge)) {
        if (Test-Path $base) {
            Get-ChildItem $base -Directory -ErrorAction SilentlyContinue |
                ForEach-Object {
                    foreach ($sub in @(
                        "Cache",
                        "Code Cache",
                        "GPUCache",
                        "Service Worker\CacheStorage"
                    )) {
                        $path = Join-Path $_.FullName $sub
                        if (Test-Path $path) {
                            Remove-FilesSafely $path
                        }
                    }
                }
        }
    }
}

if ($CleanPackageCache) {
    if (Get-Command python -ErrorAction SilentlyContinue) {
        python -m pip cache dir
        python -m pip cache purge
    }

    if (Get-Command uv -ErrorAction SilentlyContinue) {
        uv cache clean --dry-run
    }

    if (Get-Command npm -ErrorAction SilentlyContinue) {
        npm cache verify
    }
}
